document.addEventListener("DOMContentLoaded", () => { const chatForm = document.getElementById("chatForm"); const input = document.getElementById("userInput"); const messages = document.getElementById("messages"); const chatArea = document.getElementById("chatArea"); const sendBtn = document.getElementById("sendBtn"); const newChatBtn = document.getElementById("newChatBtn"); const historyList = document.getElementById("historyList"); let currentConversation = []; let allSessions = JSON.parse(localStorage.getItem("smartChatSessions")) || []; renderHistoryUI(); if (newChatBtn) { newChatBtn.addEventListener("click", () => { currentConversation = []; messages.innerHTML = `

👋 Welcome to SmartChat AI

Hello! Ask me anything or try: "Design a logo for Toynix"

`; if (input) { input.value = ""; input.focus(); } if (chatArea) chatArea.scrollTop = 0; }); } if (chatForm) { chatForm.addEventListener("submit", async (e) => { e.preventDefault(); const text = input.value.trim(); if (!text) return; addMessageToDOM(text, "user"); input.value = ""; if (window.innerWidth <= 768) input.blur(); const aiDiv = addMessageToDOM("🎨 Processing your request...", "ai"); if (sendBtn) sendBtn.disabled = true; try { const res = await fetch("/api/chat", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ message: text }) }); const data = await res.json(); if (res.ok) { if (data.type === "image" && data.imageUrl) { // 🎨 Render Logo Image with Download Button aiDiv.innerHTML = `

${data.answer}

📥 Download Logo
`; currentConversation.push({ question: text, answer: "[Generated Logo Image]" }); } else { // 💬 Render Normal Text const answerText = data.answer || "No response"; aiDiv.innerHTML = `

${answerText}

`; currentConversation.push({ question: text, answer: answerText }); } saveToRecentChats(); } else { aiDiv.innerHTML = `

⚠️ ${data.error || "API Error"}

`; } } catch (err) { aiDiv.innerHTML = `

❌ Connection Error: ${err.message}

`; } finally { if (sendBtn) sendBtn.disabled = false; if (chatArea) chatArea.scrollTop = chatArea.scrollHeight; } }); } function addMessageToDOM(text, type) { const div = document.createElement("div"); div.className = `message ${type === "user" ? "user-message" : "ai-message"}`; div.innerHTML = `

${text}

`; messages.appendChild(div); if (chatArea) chatArea.scrollTop = chatArea.scrollHeight; return div; } function saveToRecentChats() { if (currentConversation.length === 0) return; let existingIndex = allSessions.findIndex(session => session.length > 0 && session[0].question === currentConversation[0].question ); if (existingIndex !== -1) { allSessions[existingIndex] = [...currentConversation]; } else { allSessions.unshift([...currentConversation]); } allSessions = allSessions.slice(0, 15); localStorage.setItem("smartChatSessions", JSON.stringify(allSessions)); renderHistoryUI(); } function renderHistoryUI() { if (!historyList) return; historyList.innerHTML = ""; if (allSessions.length === 0) { historyList.innerHTML = `
  • No recent chats
    • `; return; } allSessions.forEach((session) => { if (!session || session.length === 0) return; const li = document.createElement("li"); li.textContent = session[0].question; li.onclick = () => { messages.innerHTML = ""; currentConversation = [...session]; currentConversation.forEach(item => { addMessageToDOM(item.question, "user"); addMessageToDOM(item.answer, "ai"); }); }; historyList.appendChild(li); }); } });
